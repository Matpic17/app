/* * Shared logic — loaded by the dashboard (browser) AND by the server (PDF, export), * so that the numbers on screen, in the PDF and in the export are always the same. * Plain JavaScript only: no HTML tags in this file. */ var LOGIC = (function () { 'use strict'; /* ---------- Definitions shown on the indicators ---------- */ var KPI = { open: { title: 'Open cases', unit: 'cases', desc: 'Cases where this programme is still open (cancelled, closed and solved cases excluded).' }, decision: { title: 'Solutions awaiting decision', unit: 'solutions', desc: 'Proposed solutions + groups with no solution at all, on open cases.' }, applied: { title: 'Applied solutions still blocking', unit: 'solutions', desc: 'Applied solutions whose status is not closed yet (X/Y).' } }; var CAT_TITLE = {}; SUB.decision.concat(SUB.applied).forEach(function (s) { CAT_TITLE[s.id] = s.title; }); var MISSING_KEY = '__MISSING__', NONE_KEY = '__NONE__'; /* ---------- Filters (OG, case leader, supplier) ---------- */ function matchFilters(d, f) { f = f || {}; var ogs = f.ogs || []; return (!ogs.length || ogs.indexOf(d.og) !== -1) && (!f.leader || d.leader === f.leader) && (!f.supplier || d.supplier === f.supplier); } function filtersActive(f) { return !!(f && ((f.ogs && f.ogs.length) || f.leader || f.supplier)); } function filterLabel(f) { var parts = []; if (f && f.ogs && f.ogs.length) parts.push('OG: ' + f.ogs.join(', ')); if (f && f.leader) parts.push('Case leader: ' + f.leader); if (f && f.supplier) parts.push('Supplier: ' + f.supplier); return parts.join(' · '); } /* ---------- Items of a case ---------- */ // kind: blocking (applied, still open) | proposal | missing (group without any solution) | closed (applied, closed) function itemsOf(d) { var out = []; d.groups.forEach(function (g) { g.blocking.forEach(function (s) { out.push({ kind: 'blocking', cat: s.cat, type: s.type, group: g.name, sol: s }); }); g.proposals.forEach(function (s) { out.push({ kind: 'proposal', cat: 'PROPOSED', type: s.type, group: g.name, sol: s }); }); if (g.missing) out.push({ kind: 'missing', cat: 'MISSING', type: '', group: g.name, sol: null }); g.done.forEach(function (s) { out.push({ kind: 'closed', cat: 'CLOSED', type: s.type, group: g.name, sol: s }); }); }); return out; } /* Key used by the "by solution type" breakdown (missing items are listed by group). */ function breakdownKey(it, sel) { if (it.kind === 'missing') return sel && sel.sub === 'MISSING' ? 'All open cases' : KPI[sel.kpi].title]; if (sel.sub) parts.push(CAT_TITLE[sel.sub] || sel.sub); if (sel.type) parts.push(keyLabel(sel.type)); return parts.join(' › '); } /* ---------- Indicators ---------- */ function computeKpis(dossiers, closedCount) { var k = { open: dossiers.length, closed: closedCount || 0, nNew: 0, proposed: 0, missing: 0, applied: 0, cats: { NO_IMPL: 0, IMPL_CLOSED: 0, IN_PROGRESS: 0, MANUAL: 0, UNCLASSIFIED: 0, PROPOSED: 0, MISSING: 0 } }; dossiers.forEach(function (d) { if (d.isNew) k.nNew++; itemsOf(d).forEach(function (it) { if (it.kind === 'blocking') { k.applied++; k.cats[it.cat] = (k.cats[it.cat] || 0) + 1; } else if (it.kind === 'proposal') { k.proposed++; k.cats.PROPOSED++; } else if (it.kind === 'missing') { k.missing++; k.cats.MISSING++; } }); }); k.decision = k.proposed + k.missing; k.total = k.open + k.closed; k.rate = k.total ? x.n > 0 : true; }) .sort(function (a, b) { return (b.n - a.n) || String(a.d.id).localeCompare(String(b.d.id), undefined, { numeric: true }); }) .map(function (x) { return x.d; }); return { filtered: filtered, closed: closed, selected: selected, kpi: computeKpis(filtered, closed.length) }; } return { KPI: KPI, SUB: SUB, CAT_TITLE: CAT_TITLE, MISSING_KEY: MISSING_KEY, matchFilters: matchFilters, filtersActive: filtersActive, filterLabel: filterLabel, itemsOf: itemsOf, itemMatches: itemMatches, breakdownKey: breakdownKey, keyLabel: keyLabel, selectionLabel: selectionLabel, computeKpis: computeKpis, breakdown: breakdown, annotate: annotate, scope: scope }; })();
